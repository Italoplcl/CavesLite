import dev.caveslite.herobrine.GroundRoute;
import dev.caveslite.herobrine.GroundRoute.Cell;
import java.util.*;
public class VerifyHerobrineRoute {
    static int checks;
    static void reaches(List<Cell> cells, Cell start, Cell goal) {
        Set<Cell> allowed=new HashSet<>(cells);Cell position=start;
        for(int i=0;i<40 && !position.equals(goal);i++) {
            Cell next=GroundRoute.next(position,goal,allowed::contains,(a,b)->true);
            if(next==null)throw new AssertionError("Route stopped at "+position+" toward "+goal);
            if(Math.abs(next.x()-position.x())+Math.abs(next.z()-position.z())!=1)throw new AssertionError("Skipped wall/cell");
            position=next;
        }
        if(!position.equals(goal))throw new AssertionError("Did not reach "+goal);checks++;
    }
    public static void main(String[] args) {
        List<Cell> stairs=List.of(new Cell(0,0,0),new Cell(1,1,0),new Cell(2,2,0),new Cell(3,3,0));
        reaches(stairs,stairs.getFirst(),stairs.getLast());
        reaches(stairs,stairs.getLast(),stairs.getFirst());
        reaches(List.of(new Cell(0,0,0),new Cell(0,1,1),new Cell(1,2,1),new Cell(1,3,0),new Cell(0,4,0)),new Cell(0,0,0),new Cell(0,4,0));
        reaches(List.of(new Cell(0,0,0),new Cell(0,0,1),new Cell(1,0,1),new Cell(2,0,1),new Cell(2,0,0)),new Cell(0,0,0),new Cell(2,0,0));
        reaches(List.of(new Cell(0,3,0),new Cell(1,0,0)),new Cell(0,3,0),new Cell(1,0,0));
        Cell start=new Cell(0,0,0),goal=new Cell(1,2,0);
        if(GroundRoute.next(start,goal,c->c.equals(goal),(a,b)->true)!=null)throw new AssertionError("Climbed two-block wall");checks++;
        if(GroundRoute.next(start,new Cell(1,-4,0),c->true,(a,b)->false)!=null)throw new AssertionError("Crossed blocked transition");checks++;
        if(GroundRoute.next(start,goal,c->false,(a,b)->true)!=null)throw new AssertionError("Entered unsafe/unloaded terrain");checks++;
        System.out.println("PASS: "+checks+" Herobrine routing scenarios (up/down stairs, stacked floors, detour, drop, blocked and unsafe terrain)");
    }
}
